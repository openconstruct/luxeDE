cmake_minimum_required(VERSION 3.25)
project(luxe_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(X11 QUIET)
if(X11_FOUND)
add_executable(run-nested-x run-nested-x.cc)
target_link_libraries(run-nested-x PRIVATE X11::X11)
endif()
add_executable(check-source check-source.cc)
find_package(LibArchive QUIET)
if(LibArchive_FOUND)
  add_executable(package-tools package-tools.cc)
  target_link_libraries(package-tools PRIVATE LibArchive::LibArchive)
endif()
include(CTest)
if(BUILD_TESTING AND TARGET package-tools)
  add_test(NAME native-package-tools COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/test-packaging.sh" "$<TARGET_FILE:package-tools>")
endif()
if(BUILD_TESTING)
  add_test(NAME native-source-hygiene COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/test-source.sh" "$<TARGET_FILE:check-source>")
endif()
add_executable(deduplicate-icons deduplicate-icons.cc)
find_package(nlohmann_json QUIET)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(DIAG_PIXBUF QUIET IMPORTED_TARGET gdk-pixbuf-2.0)
endif()
find_path(DIAG_XRES_INCLUDE X11/extensions/XRes.h)
find_library(DIAG_XRES_LIBRARY NAMES XRes)
if(nlohmann_json_FOUND AND TARGET PkgConfig::DIAG_PIXBUF AND X11_FOUND AND DIAG_XRES_INCLUDE AND DIAG_XRES_LIBRARY)
  add_executable(diagnostics diagnostics.cc)
  target_include_directories(diagnostics PRIVATE "${DIAG_XRES_INCLUDE}")
  target_link_libraries(diagnostics PRIVATE nlohmann_json::nlohmann_json PkgConfig::DIAG_PIXBUF X11::X11 "${DIAG_XRES_LIBRARY}" X11::Xext)
endif()
if(BUILD_TESTING AND TARGET diagnostics)
  add_executable(test-diagnostics test-diagnostics.cc)
  target_link_libraries(test-diagnostics PRIVATE nlohmann_json::nlohmann_json)
  add_test(NAME native-diagnostics COMMAND test-diagnostics "$<TARGET_FILE:diagnostics>" "${CMAKE_CURRENT_SOURCE_DIR}/../..")
endif()
if(PkgConfig_FOUND)
  pkg_check_modules(FIXTURE_GIO QUIET IMPORTED_TARGET gio-2.0)
endif()
if(nlohmann_json_FOUND AND TARGET PkgConfig::FIXTURE_GIO AND TARGET PkgConfig::DIAG_PIXBUF AND X11_FOUND)
  add_executable(behavior-fixtures behavior-fixtures.cc)
  target_link_libraries(behavior-fixtures PRIVATE nlohmann_json::nlohmann_json PkgConfig::FIXTURE_GIO PkgConfig::DIAG_PIXBUF X11::X11)
endif()
if(PkgConfig_FOUND AND nlohmann_json_FOUND)
  foreach(gtk_version IN ITEMS 4 3)
    if(gtk_version EQUAL 4)
      pkg_check_modules(ICON_GTK4 QUIET IMPORTED_TARGET gtk4)
    else()
      pkg_check_modules(ICON_GTK3 QUIET IMPORTED_TARGET gtk+-3.0)
    endif()
    if(TARGET PkgConfig::ICON_GTK${gtk_version})
      add_executable(icon-fixtures-gtk${gtk_version} icon-fixtures.cc)
      target_link_libraries(icon-fixtures-gtk${gtk_version} PRIVATE nlohmann_json::nlohmann_json PkgConfig::ICON_GTK${gtk_version})
    endif()
  endforeach()
endif()
if(TARGET PkgConfig::ICON_GTK3 AND X11_FOUND)
  add_executable(gtk3-fixtures gtk3-fixtures.cc)
  target_link_libraries(gtk3-fixtures PRIVATE nlohmann_json::nlohmann_json PkgConfig::ICON_GTK3 X11::X11)
  # GtkStatusIcon intentionally exercises compatibility with real GTK3 clients.
  target_compile_options(gtk3-fixtures PRIVATE -Wno-deprecated-declarations)
  if(TARGET behavior-fixtures)
    add_dependencies(behavior-fixtures gtk3-fixtures)
  endif()
endif()
if(nlohmann_json_FOUND)
  add_executable(import-themes import-themes.cc)
  target_link_libraries(import-themes PRIVATE nlohmann_json::nlohmann_json)
endif()
if(PkgConfig_FOUND AND nlohmann_json_FOUND)
  pkg_check_modules(THEME_ADW QUIET IMPORTED_TARGET libadwaita-1>=1.4)
  foreach(version IN ITEMS 3 4)
    if(TARGET PkgConfig::ICON_GTK${version})
      add_executable(theme-widget-gtk${version} theme-widget-fixture.cc)
      target_link_libraries(theme-widget-gtk${version} PRIVATE nlohmann_json::nlohmann_json PkgConfig::ICON_GTK${version})
      target_compile_options(theme-widget-gtk${version} PRIVATE -Wno-deprecated-declarations)
      if(TARGET behavior-fixtures)
        add_dependencies(behavior-fixtures theme-widget-gtk${version})
      endif()
    endif()
  endforeach()
  if(TARGET PkgConfig::THEME_ADW)
    add_executable(theme-widget-adw theme-widget-fixture.cc)
    target_compile_definitions(theme-widget-adw PRIVATE WITH_ADWAITA)
    target_compile_options(theme-widget-adw PRIVATE -Wno-deprecated-declarations)
    target_link_libraries(theme-widget-adw PRIVATE nlohmann_json::nlohmann_json PkgConfig::THEME_ADW)
    if(TARGET behavior-fixtures)
      add_dependencies(behavior-fixtures theme-widget-adw)
    endif()
  endif()
endif()
if(nlohmann_json_FOUND)
  add_executable(panel-profile panel-profile.cc)
  target_link_libraries(panel-profile PRIVATE nlohmann_json::nlohmann_json)
endif()
