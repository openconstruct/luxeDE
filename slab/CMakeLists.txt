cmake_minimum_required(VERSION 3.25)
project(slab VERSION 0.2.9 LANGUAGES C)
set(SLAB_RELEASE "${PROJECT_VERSION}~testing6")
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "/usr" CACHE PATH "Install prefix" FORCE)
endif()
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4-x11>=4.12 gio-unix-2.0)
pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
add_executable(slab src/main.c src/pins.c)
target_compile_features(slab PRIVATE c_std_11)
target_compile_definitions(slab PRIVATE SLAB_VERSION="${SLAB_RELEASE}")
target_compile_options(slab PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(slab PRIVATE ../shared)
target_link_libraries(slab PRIVATE PkgConfig::GTK PkgConfig::X11 m)
add_executable(slab-settings src/settings.c src/pins.c)
target_compile_features(slab-settings PRIVATE c_std_11)
target_compile_options(slab-settings PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(slab-settings PRIVATE SLAB_VERSION="${SLAB_RELEASE}")
target_include_directories(slab-settings PRIVATE ../shared)
target_link_libraries(slab-settings PRIVATE PkgConfig::GTK)
find_program(GLIB_COMPILE_SCHEMAS glib-compile-schemas REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/schemas")
configure_file(data/org.luxe.slab.gschema.xml schemas/org.luxe.slab.gschema.xml COPYONLY)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/schemas/gschemas.compiled"
  COMMAND "${GLIB_COMPILE_SCHEMAS}" --strict "${CMAKE_CURRENT_BINARY_DIR}/schemas"
  DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/schemas/org.luxe.slab.gschema.xml")
add_custom_target(slab-schemas ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/schemas/gschemas.compiled")
install(TARGETS slab slab-settings RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/org.luxe.Slab.desktop data/org.luxe.SlabSettings.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/org.luxe.slab.gschema.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/glib-2.0/schemas)
install(FILES copyright README.md ChangeLog DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/slab)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
set(CPACK_PACKAGE_NAME slab)
set(CPACK_PACKAGE_VERSION "${SLAB_RELEASE}")
set(CPACK_PACKAGE_CONTACT "Jerry Howell")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "GTK4 dock for luxe (X11)")
set(CPACK_DEBIAN_PACKAGE_SECTION x11)
set(CPACK_DEBIAN_PACKAGE_PRIORITY optional)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libgtk-4-1 (>= 4.12)")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/data/triggers")
set(CPACK_PRE_BUILD_SCRIPTS "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/FixPackagePermissions.cmake")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_SET_DESTDIR ON)
set(CPACK_STRIP_FILES ON)
include(CPack)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME slab-reveal-stopped COMMAND dbus-run-session -- env DISPLAY= $<TARGET_FILE:slab> --reveal)
  set_tests_properties(slab-reveal-stopped PROPERTIES TIMEOUT 5)
  add_executable(test-slab-settings tests/test-settings.c src/pins.c)
  target_include_directories(test-slab-settings PRIVATE ../shared)
  target_link_libraries(test-slab-settings PRIVATE PkgConfig::GTK)
  add_test(NAME slab-settings COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/RunNestedX.sh" $<TARGET_FILE:test-slab-settings>)
  set_tests_properties(slab-settings PROPERTIES TIMEOUT 30 ENVIRONMENT "GSETTINGS_SCHEMA_DIR=${CMAKE_CURRENT_BINARY_DIR}/schemas")
  add_executable(test-slab tests/test-dock.c src/pins.c)
  target_compile_features(test-slab PRIVATE c_std_11)
  target_compile_definitions(test-slab PRIVATE SLAB_TESTING)
  target_include_directories(test-slab PRIVATE ../shared)
  target_link_libraries(test-slab PRIVATE PkgConfig::GTK PkgConfig::X11 m)
  add_test(NAME slab-window-model COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/RunNestedX.sh" $<TARGET_FILE:test-slab>)
  set_tests_properties(slab-window-model PROPERTIES TIMEOUT 30 ENVIRONMENT "GSETTINGS_SCHEMA_DIR=${CMAKE_CURRENT_BINARY_DIR}/schemas")
  add_executable(test-slab-pins tests/test-pins.c src/pins.c)
  target_link_libraries(test-slab-pins PRIVATE PkgConfig::GTK)
  add_test(NAME slab-pins COMMAND test-slab-pins)
  find_program(OPENBOX_BIN openbox)
  find_program(XDOTOOL_BIN xdotool)
  if(OPENBOX_BIN AND XDOTOOL_BIN)
    add_executable(test-slab-wm tests/test-wm.c src/pins.c)
    target_include_directories(test-slab-wm PRIVATE ../shared)
    target_link_libraries(test-slab-wm PRIVATE PkgConfig::GTK PkgConfig::X11 m)
    add_test(NAME slab-openbox COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/../cmake/RunNestedX.sh" $<TARGET_FILE:test-slab-wm>)
    set_tests_properties(slab-openbox PROPERTIES TIMEOUT 30 ENVIRONMENT "GSETTINGS_SCHEMA_DIR=${CMAKE_CURRENT_BINARY_DIR}/schemas")
  endif()
endif()
